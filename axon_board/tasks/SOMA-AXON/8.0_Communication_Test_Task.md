## 📋 概要

SOMA-AXON間の全コマンド通信を自動的にテストし、各コマンドの動作確認と統計情報を取得するテストタスクを実装する。

---

## 🎯 目的

1. **全コマンドの動作確認**: CHKIRQ、SETAXON、NOP、FW更新系コマンドの動作を検証
2. **通信品質の測定**: タイムアウト、CRCエラー、NACK応答の頻度を記録
3. **自動テスト**: 手動テストを自動化し、効率的な検証を実現
4. **継続的な品質保証**: リグレッションテストとして使用

---

## 📊 テスト対象コマンド

| No | コマンド | CMD_ID | 応答 | テスト内容 |
|----|---------|--------|------|-----------|
| 1 | CHKIRQ | 0x49 | ATIRQ (0x6A) | 基本ポーリング通信 |
| 2 | SETAXON | 0x4A | ACK (0x00) / NACK (0x90) | 設定変更と応答 |
| 3 | NOP | 0x50 | ACK (0x00) | 無操作コマンド |
| 4 | SETOKEY | 0x11 | ACK (0x00) / NACK (0x90) | 運用鍵設定（基本） |
| 5 | AFWUP | 0x18 | ACK (0x00) / NACK (0x90) | FW更新要求 |
| 6 | CODEPKT | 0xA5 | CODEOK (0xA6) / CODENG (0xA7) | FWコード受信 |
| 7 | ERRCHK | 0xC4 | CODEFIN (0xC5) | FWエラーチェック |

## 🏗️ 実装設計

### テストタスク構造

```c
typedef struct {
    uint32_t total_tests;        // 総テスト数
    uint32_t passed_tests;       // 成功数
    uint32_t failed_tests;       // 失敗数
    uint32_t timeout_count;      // タイムアウト数
    uint32_t crc_error_count;    // CRCエラー数
    uint32_t nack_count;         // NACK応答数
    uint32_t start_time_ms;      // テスト開始時刻
    uint32_t end_time_ms;        // テスト終了時刻
} test_stats_t;
```

### テストシーケンス

```
1. 初期化
   - UART通信確立
   - ポート選択（Port 1-2）
   - 統計情報リセット

2. TC-01: CHKIRQ/ATIRQ
   - CHKIRQ送信
   - ATIRQ受信
   - STATUS/CRC検証

3. TC-02: SETAXON/ACK
   - 正常値でSETAXON送信
   - ACK受信確認
   - 不正値でNACK確認

4. TC-03: NOP
   - NOP送信
   - ACK受信確認

5. TC-04: SETOKEY
   - 鍵データ送信
   - ACK受信確認

6. TC-05: AFWUP
   - バージョン情報送信
   - ACK/NACK受信確認

7. TC-06: CODEPKT
   - FWコードデータ送信
   - CODEOK/CODENG受信確認

8. TC-07: ERRCHK
   - エラーチェック要求
   - CODEFIN受信確認

9. 統計出力
   - 成功率
   - エラー内訳
   - 実行時間
```

## 💻 実装コード例 (SOMA側)

### test/soma_axon_test.c

```c
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "esp_log.h"
#include "../protocol/src/s2a_packet.h"

static const char* TAG = "AXON_TEST";

// テスト統計
static test_stats_t g_test_stats = {0};

// ヘルパー関数
static bool send_and_receive(uint8_t port, uint8_t* tx_frame, uint8_t* rx_frame, uint32_t timeout_ms);
static void record_test_result(bool passed, const char* test_name);
static uint16_t crc16_calculate(const uint8_t* data, size_t len);

// TC-01: CHKIRQ/ATIRQ テスト
static bool test_chkirq_atirq(uint8_t port) {
    ESP_LOGI(TAG, "[TC-01] CHKIRQ/ATIRQ Test (Port %d)", port);
    
    uint8_t tx_frame[36] = {0};
    tx_frame[0] = 0x14;  // Header
    tx_frame[1] = 0x20;  // Length
    tx_frame[2] = 0x49;  // CHKIRQ
    
    uint16_t crc = crc16_calculate(tx_frame, 34);
    tx_frame[34] = (uint8_t)(crc & 0xFF);
    tx_frame[35] = (uint8_t)((crc >> 8) & 0xFF);
    
    uint8_t rx_frame[36] = {0};
    if (!send_and_receive(port, tx_frame, rx_frame, 100)) {
        return false;
    }
    
    // 応答検証
    bool valid = (rx_frame[0] == 0x10) && (rx_frame[2] == 0x6A);
    
    if (valid) {
        uint16_t status = rx_frame[8] | (rx_frame[9] << 8);
        ESP_LOGI(TAG, "STATUS: 0x%04X", status);
    }
    
    return valid;
}

// TC-02: SETAXON/ACK テスト
static bool test_setaxon_ack(uint8_t port) {
    ESP_LOGI(TAG, "[TC-02] SETAXON/ACK Test (Port %d)", port);
    
    uint8_t tx_frame[36] = {0};
    tx_frame[0] = 0x14;
    tx_frame[1] = 0x20;
    tx_frame[2] = 0x4A;  // SETAXON
    tx_frame[3] = 3;     // FACE_N
    tx_frame[4] = 5;     // SET_FACE_N
    tx_frame[5] = 3;     // CASH_VLU LSB (300円)
    tx_frame[6] = 0;     // CASH_VLU MSB
    tx_frame[7] = 0x01;  // SOL ON
    tx_frame[8] = 0x07;  // LED RGB
    tx_frame[9] = 0x05;  // TIMEOUT 45sec
    
    uint16_t crc = crc16_calculate(tx_frame, 34);
    tx_frame[34] = (uint8_t)(crc & 0xFF);
    tx_frame[35] = (uint8_t)((crc >> 8) & 0xFF);
    
    uint8_t rx_frame[36] = {0};
    if (!send_and_receive(port, tx_frame, rx_frame, 100)) {
        return false;
    }
    
    return (rx_frame[0] == 0x10) && (rx_frame[2] == 0x00);
}

// メインテストタスク
void soma_axon_communication_test_task(void* arg) {
    ESP_LOGI(TAG, "=== SOMA-AXON Communication Test Started ===");
    
    uint8_t target_port = 1;
    g_test_stats.start_time_ms = xTaskGetTickCount() * portTICK_PERIOD_MS;
    
    vTaskDelay(pdMS_TO_TICKS(1000));
    
    // テスト実行
    record_test_result(test_chkirq_atirq(target_port), "TC-01: CHKIRQ/ATIRQ");
    vTaskDelay(pdMS_TO_TICKS(100));
    
    record_test_result(test_setaxon_ack(target_port), "TC-02: SETAXON/ACK");
    vTaskDelay(pdMS_TO_TICKS(100));
    
    // ... 他のテストケース ...
    
    g_test_stats.end_time_ms = xTaskGetTickCount() * portTICK_PERIOD_MS;
    
    // 統計出力
    ESP_LOGI(TAG, "=== Test Summary ===");
    ESP_LOGI(TAG, "Total: %lu", g_test_stats.total_tests);
    ESP_LOGI(TAG, "Passed: %lu (%.1f%%)", 
             g_test_stats.passed_tests,
             (float)g_test_stats.passed_tests / g_test_stats.total_tests * 100.0f);
    ESP_LOGI(TAG, "Failed: %lu", g_test_stats.failed_tests);
    ESP_LOGI(TAG, "Timeouts: %lu", g_test_stats.timeout_count);
    ESP_LOGI(TAG, "CRC Errors: %lu", g_test_stats.crc_error_count);
    ESP_LOGI(TAG, "NACK Count: %lu", g_test_stats.nack_count);
    ESP_LOGI(TAG, "Duration: %lu ms", 
             g_test_stats.end_time_ms - g_test_stats.start_time_ms);
    
    vTaskDelete(NULL);
}
```

### 使用方法

#### AXON側（受信側）

```c
// テストモード初期化（起動時に自動実行）
axon_test_init();

// テストモード有効化（デフォルト設定）
axon_test_enable(NULL);

// カスタム設定でテストモード有効化
axon_test_config_t config = {
    .enable_test_mode = true,
    .log_all_commands = true,     // 全コマンドをログ出力
    .force_nack_setaxon = false,  // SETAXON強制NACK（テスト用）
    .simulate_timeout = false,    // タイムアウトシミュレーション
    .verbose_status = true        // STATUS詳細表示
};
axon_test_enable(&config);

// 統計表示
axon_test_print_stats();

// 統計リセット
axon_test_reset_stats();

// テストモード無効化
axon_test_disable();
```
